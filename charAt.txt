const arrayNombre = ["hector", "Javier"];


function cuadraNombres(params) {

  let dato = [];

  for (let i = 0; i < params.length; i++) {

    let palabra = params[i];

    dato.push(palabra.charAt(0).toUpperCase() + palabra.slice(1))


  }

  return dato;

}

console.log(cuadraNombres(arrayNombre));
